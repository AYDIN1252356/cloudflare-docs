from telegram import Update, InlineQueryResultArticle, InputTextMessageContent
from telegram.ext import Updater, CommandHandler, CallbackContext, InlineQueryHandler, Filters, MessageHandler
from datetime import datetime
import uuid

# تاریخ کنکور
EXAM_DATE = datetime(2025, 5, 1, 7, 0, 0)  # 12 اردیبهشت 1404 ساعت 7 صبح

def days_until_exam(update: Update, context: CallbackContext) -> None:
    """محاسبه تعداد روزهای باقی‌مانده تا کنکور و ارسال آن به کاربر."""
    today = datetime.now()
    remaining_time = EXAM_DATE - today
    remaining_days = remaining_time.days
    remaining_seconds = remaining_time.seconds

    if remaining_time.total_seconds() > 0:
        hours, remainder = divmod(remaining_seconds, 3600)
        minutes, seconds = divmod(remainder, 60)
        message = (
            f"تا کنکور {remaining_days} روز، {hours} ساعت، {minutes} دقیقه و {seconds} ثانیه باقی مانده است. 📚"
        )
    elif remaining_time.total_seconds() == 0:
        message = "امروز روز کنکور است! موفق باشی! 🌟"
    else:
        message = "کنکور گذشته است. امیدوارم موفق شده باشی! ✅"

    update.message.reply_text(message)

def inline_query(update: Update, context: CallbackContext) -> None:
    """پاسخ به درخواست اینلاین."""
    query = update.inline_query.query
    if query == "":
        return

    today = datetime.now()
    remaining_time = EXAM_DATE - today
    remaining_days = remaining_time.days
    remaining_seconds = remaining_time.seconds

    if remaining_time.total_seconds() > 0:
        hours, remainder = divmod(remaining_seconds, 3600)
        minutes, seconds = divmod(remainder, 60)
        result_message = (
            f"تا کنکور {remaining_days} روز، {hours} ساعت، {minutes} دقیقه و {seconds} ثانیه باقی مانده است. 📚"
        )
    elif remaining_time.total_seconds() == 0:
        result_message = "امروز روز کنکور است! موفق باشی! 🌟"
    else:
        result_message = "کنکور گذشته است. امیدوارم موفق شده باشی! ✅"

    results = [
        InlineQueryResultArticle(
            id=str(uuid.uuid4()),
            title="روزشمار کنکور",
            input_message_content=InputTextMessageContent(result_message),
        )
    ]

    update.inline_query.answer(results)

def start(update: Update, context: CallbackContext) -> None:
    """پاسخ به فرمان /start."""
    update.message.reply_text(
        "سلام! 👋\nبرای دیدن تعداد روزهای باقی‌مانده تا کنکور، دستور /countdown را ارسال کن، یا در گروه از حالت اینلاین استفاده کن."
    )

def custom_days_command(update: Update, context: CallbackContext) -> None:
    """مدیریت دستورات سفارشی برای روزشمار کنکور."""
    valid_commands = ["/روزشمار", "/روز شمار", "/روزشمار کنکور", "/روزشمارکنکور", "/روز شمار کنکور"]
    if update.message.text.strip() in valid_commands:
        days_until_exam(update, context)

def main() -> None:
    """اجرای ربات تلگرام."""
    # توکن ربات خود را اینجا وارد کنید
    TOKEN = "YOUR_TELEGRAM_BOT_TOKEN"

    updater = Updater(TOKEN)

    # اضافه کردن فرمان‌ها
    updater.dispatcher.add_handler(CommandHandler("start", start))
    updater.dispatcher.add_handler(CommandHandler("countdown", days_until_exam))
    updater.dispatcher.add_handler(InlineQueryHandler(inline_query))
    updater.dispatcher.add_handler(MessageHandler(Filters.text & Filters.group, custom_days_command))

    # شروع ربات
    updater.start_polling()
    updater.idle()

if __name__ == "__main__":
    main()
